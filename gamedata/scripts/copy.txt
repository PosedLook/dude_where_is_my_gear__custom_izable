local _oldsoul_OnDeath = soulslike_scenarios.SoulslikeScenarioLogic.OnDeath
function soulslike_scenarios.SoulslikeScenarioLogic:OnDeath()
    dwimg_utils.print_dbg("Overwritten!!")
    if dwimg_mcm.get_config("main_settings/soulslike_overwrite") == 1 and dwimg_mcm.get_config("main_settings/on_off") == 2 and (dwimg_mcm.get_config("main_settings/use_soulslike_ambush") == false and true) then
        dwimg_utils.print_dbg("Cancelled!!")
        return
    else
        dwimg_utils.print_dbg("Kept!!")
        _oldsoul_OnDeath(self)
    end
end
